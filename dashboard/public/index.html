const express = require("express"); const path = require("node:path"); const { getGuildSettings, getOpenTickets, listResponders } = require("../db"); const app = express(); const port = process.env.PORT || 3000; app.use(express.json()); app.use(express.static(path.join(__dirname, "public"))); app.get("/", (req, res) => { res.sendFile(path.join(__dirname, "public", "index.html")); }); app.get("/api/settings", (req, res) => { const guildId = req.query.guildId; if (!guildId) return res.status(400).json({ error: "guildId is required" }); res.json(getGuildSettings(guildId)); }); app.get("/api/tickets", (req, res) => { const guildId = req.query.guildId; if (!guildId) return res.status(400).json({ error: "guildId is required" }); res.json(getOpenTickets(guildId)); }); app.get("/api/autoresponders", (req, res) => { const guildId = req.query.guildId; if (!guildId) return res.status(400).json({ error: "guildId is required" }); res.json(listResponders(guildId)); }); app.listen(port, () => { console.log(`Mochii dashboard running on http://localhost:${port}`); });
